let allPerfumes = []; document.addEventListener('DOMContentLoaded', () => { // Yılı güncelle (Hata vermemesi için kontrol ekli) const yearEl = document.getElementById('currentYear'); if(yearEl) yearEl.textContent = new Date().getFullYear(); loadPerfumes(); }); async function loadPerfumes() { try { const response = await fetch('data.json'); // Eğer dosya okunamazsa burada durur ama siteyi çökertmez if (!response.ok) return; allPerfumes = await response.json(); renderPerfumes(allPerfumes); // Arama kutusunu aktif et const searchInput = document.getElementById('searchInput'); if(searchInput) { searchInput.addEventListener('input', (e) => { const term = e.target.value.toLowerCase().trim(); const filtered = allPerfumes.filter(p => p.name.toLowerCase().includes(term) || p.code.toLowerCase().includes(term) || (p.keywords && p.keywords.toLowerCase().includes(term)) ); renderPerfumes(filtered); }); } } catch (err) { console.error("Yükleme hatası, ama site çalışmaya devam ediyor."); } } function renderPerfumes(list) { const container = document.getElementById('productsContainer'); if (!container) return; if (list.length === 0) { container.innerHTML = '

Ürün bulunamadı.

'; return; } container.innerHTML = list.map(p => `

${p.name}

Kod: ${p.code}

${p.description || ''}

Ürüne Git
`).join(''); }
